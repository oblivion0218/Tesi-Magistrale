% === SLIDE 2: CONTESTO E OBIETTIVO ===
\begin{frame}{Thesis Context and Objective}
    \begin{block}{Subcritical System}
        A reactor with $k_{eff} < 1$ requires an \textbf{external neutron source} to sustain the chain reaction. During operational transients, \textbf{feedback} modify $k_{eff}$ .
    \end{block}
    
    \vspace{0.5cm}
    
    \begin{alertblock}{Objective}
        Develop a methodology to \textbf{maximize} $k_{eff}^{HFP}$ (deeply linked to power production) ensuring that $k_{eff}^{CZP} \leq 1$ under any operating condition.
    \end{alertblock}
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \item \textbf{CZP} (Cold Zero Power): $T = 300$ K — reactor shutdown
        \item \textbf{HFP} (Hot Full Power): $T_{fuel} = 600$ K, $T_{mod} = 350$ K — nominal operation 
    \end{itemize}
\end{frame}

% === SLIDE 4: FEEDBACK DI REATTIVITÀ ===
\begin{frame}{Reactivity Feedback}

    During the transition from HFP to CZP, the temperature drop induces a \textbf{reactivity insertion}. The primary feedback mechanisms are: \\
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            \textbf{Doppler Effect}
            \begin{itemize}
                \item Narrowing of $^{238}$U capture resonances due to reduced thermal agitation
                \item Decreased epithermal absorption $\Rightarrow$ \textcolor{red}{$\Delta\rho > 0$}
            \end{itemize}
            
            \vspace{0.5cm}
            
            \textbf{Moderator Effect}
            \begin{itemize}
                \item Increase in water density as temperature decreases
                \item Enhanced neutron thermalization $\Rightarrow$ \textcolor{red}{$\Delta\rho > 0$}
            \end{itemize}
        \end{column}
        
        \begin{column}{0.4\textwidth}
            \begin{alertblock}{Reactivity Variation}
                \begin{equation*}
                    \Delta\rho = \frac{k_2 - k_1}{k_1 \cdot k_2} \quad \text{ [pcm]}
                \end{equation*}
            \end{alertblock}
            
            \vspace{0.3cm}
            
            \begin{block}{Safety Constraint}
                \begin{equation*}
                    k_{max}^{HFP} \approx  k_{max}^{CZP} - \sum_i \Delta\rho_i
                \end{equation*}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% === SLIDE 9: VELENI - TEORIA ===
\begin{frame}{Neutron Poisons: Xe-135 and Sm-149}
    Other than the thermal feedback, the presence of neutron poisons (fission products with high absorption cross section) can significantly reduce $k_{eff}$ during operation. The main poisons in a uranium-fueled reactor are:
    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Xenon-135}
            \begin{itemize}
                \item $\sigma_a \approx 2 \times 10^6$ b
                \item Transient: \textbf{post-shutdown peak}
                \item Decays with $T_{1/2} = 9.2$ h
            \end{itemize}
            
            \begin{equation*}
                N_{Xe}^{eq} = \frac{(\gamma_I + \gamma_{Xe}) \Sigma_f \phi}{\lambda_{Xe} + \sigma_a \phi}
            \end{equation*}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \textbf{Samarium-149}
            \begin{itemize}
                \item $\sigma_a \approx 4.1 \times 10^4$ b
                \item \textbf{Stable} ($\lambda_{Sm} = 0$)
                \item Permanent asymptotic buildup
            \end{itemize}
            
            \begin{equation*}
                N_{Sm}^{eq} = \frac{\gamma_{Pm} \Sigma_f}{\sigma_{a,Sm}}
            \end{equation*}
            
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Performance Parameters: The Q-Factor}
    
    In an ADS system, the energy gain factor $Q$ describes the amplification of the source power within the subcritical core:
    \begin{equation*}
        Q = \frac{P_{thermal}}{P_{beam}} = \psi \frac{k_{eff}}{1 - k_{eff}}
    \end{equation*}
    Where $\psi$ represents the relative importance factor of source neutrons.
    
    For a consistent evaluation, a system $Q$-factor ($Q_{sys}$) is defined with respect to the electrical power of the synchrotron:
    \begin{equation*}
        Q_{sys} = \frac{P_{thermal}}{P_{el, sync} / \eta_{th}}
    \end{equation*}

\begin{frame}{Depletion Simulations (Burn-up)}
    \begin{block}{Transport-Isotopic Evolution Coupling}
        The simulation in \textit{depletion} mode allows tracking the time evolution of the nuclear fuel isotopic composition under the effect of the neutron flux, by solving the coupled Monte Carlo transport and Bateman equations.
    \end{block}

    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Transport Phase (OpenMC):} Calculation of the spatial and spectral neutron flux $\Phi(\vec{r}, E)$ and microscopic reaction rates ($\sigma_i \phi$) at the beginning of the time-step.
        \item \textbf{Evolution Phase (Bateman):} Use of the extracted rates to numerically solve the system of linearized differential equations, updating the atomic densities $N_i(t)$ for the next step.
        \item \textbf{Composition Feedback:} The accumulation of fission products (poisons) and the consumption of fissile material alter the macroscopic properties of the medium ($\Sigma_a, \Sigma_f$), modifying the neutron response in subsequent steps.
    \end{itemize}
\end{frame}

% --- SLIDE 1: Formulation and Operators ---
\begin{frame}{Mathematical Foundation: The Transport Equation}
The steady-state neutron balance is governed by the transport equation:

\begin{alertblock}{Operator Formulation}
\begin{equation*}
\mathcal{M} \Phi(\vec{r}, E, \hat{\Omega}) = \frac{1}{k_{eff}} \mathcal{F} \Phi(\vec{r}, E, \hat{\Omega})
\end{equation*}
\end{alertblock}

The two operators describe the competing physical rates in the core:

\begin{itemize}
\item \textbf{$\mathcal{M}$ (Migration/Destruction Operator):} Quantifies net losses. It includes geometric leakage (\textit{streaming}, $\hat{\Omega} \cdot \nabla \Phi$), total interactions ($\Sigma_t \Phi$), and kinematic repopulation due to scattering (\textit{in-scattering}).
\item \textbf{$\mathcal{F}$ (Production Operator):} Quantifies the steady-state source, describing the emission of new prompt neutrons induced by the integral fission rate ($\chi \nu \Sigma_f \Phi$).
\end{itemize}
\end{frame}

% --- SLIDE 2: Spectrum and Physical Meaning ---
\begin{frame}{Mathematical Foundation: The Krein-Rutman Theorem}
Rewriting the equation as $\mathcal{M}^{-1}\mathcal{F} \Phi = k_{eff} \Phi$, the operator $\mathcal{M}^{-1}\mathcal{F}$ maps one neutron generation to the next. Since atomic densities and cross-sections are strictly positive quantities, this operator is \textbf{rigorously positive}.

\begin{block}{Eigenvalue Dominance ($k_{eff}$)}
By the \textbf{Krein-Rutman Theorem}, a positive integral operator guarantees the existence of a fundamental eigenvalue ($k_0 \equiv k_{eff}$) that is real, positive, and \textbf{larger in magnitude} than all other eigenvalues in the spectrum ($|k_0| > |k_n|$).
\end{block}

\begin{alertblock}{Physical Meaning of the Eigenfunction ($\Phi_0$)}
Only the dominant eigenvalue $k_{eff}$ is associated with an eigenfunction $\Phi_0$ that is \textbf{everywhere positive}. Higher-order harmonics inevitably exhibit nodes (negative fluxes, which are physically impossible in steady-state conditions).
\end{alertblock}
\end{frame}
