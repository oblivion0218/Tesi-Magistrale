% === SLIDE 2: CONTESTO E OBIETTIVO ===
\begin{frame}{Contesto e Obiettivo della Tesi}
    \begin{block}{Sistema Sottocritico}
        Un reattore con $k_{eff} < 1$ richiede una \textbf{sorgente esterna} di neutroni. La potenza è proporzionale all'intensità della sorgente, ma i \textbf{feedback} modificano $k_{eff}$ durante i transitori.
    \end{block}
    
    \vspace{0.5cm}
    
    \begin{alertblock}{Obiettivo}
        Sviluppare una metodologia per \textbf{massimizzare} $k_{eff}^{HFP}$ garantendo che $k_{eff}^{CZP} \leq 1$ in ogni condizione operativa.
    \end{alertblock}
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \item \textbf{CZP} (Cold Zero Power): $T = 300$ K — reattore spento
        \item \textbf{HFP} (Hot Full Power): $T_{fuel} = 600$ K, $T_{mod} = 350$ K — operatività nominale
    \end{itemize}
\end{frame}

\begin{frame}{Progetto SYNERGY: Sorgente Fotonica}

                \item Rate fotonico: $S_\gamma = 2.02 \times 10^{17}$ $\gamma$/s

% === SLIDE 4: FEEDBACK DI REATTIVITÀ ===
\begin{frame}{Feedback di Reattività}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Effetto Doppler}
            \begin{itemize}
                \item Broadening delle risonanze di cattura dell'$^{238}$U
                \item Aumenta le catture $\Rightarrow$ \textcolor{red}{$\Delta\rho < 0$}
            \end{itemize}
            
            \vspace{0.5cm}
            
            \textbf{Effetto Moderatore}
            \begin{itemize}
                \item Riduzione densità acqua con $T$
                \item Degrada rallentamento $\Rightarrow$ \textcolor{red}{$\Delta\rho < 0$}
            \end{itemize}
        \end{column}
        
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Variazione di Reattività}
                \begin{equation*}
                    \Delta\rho = \frac{k_2 - k_1}{k_1 \cdot k_2}
                \end{equation*}
            \end{alertblock}
            
            \vspace{0.3cm}
            
            \begin{block}{Vincolo di Sicurezza}
                \begin{equation*}
                    k_{max}^{HFP} = 1 - \sum_i |\Delta\rho_i|
                \end{equation*}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
